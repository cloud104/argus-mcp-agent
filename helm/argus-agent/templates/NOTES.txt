🚀 Argus Agent has been deployed!

1. Get the application URL:
{{- if .Values.ingress.enabled }}
  https://{{ .Values.ingress.host }}

  API:        https://{{ .Values.ingress.host }}/api
  MCP Server: https://{{ .Values.ingress.host }}/mcp
  UI:         https://{{ .Values.ingress.host }}/
{{- else if contains "LoadBalancer" .Values.api.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  You can watch the status by running:

  kubectl get svc --namespace {{ .Values.global.namespace }} -w {{ include "argus-agent.fullname" . }}-api

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Values.global.namespace }} {{ include "argus-agent.fullname" . }}-api -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  echo "Argus Agent URL: http://$SERVICE_IP:{{ .Values.api.service.port }}"
{{- else }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Values.global.namespace }} -l "app.kubernetes.io/name={{ include "argus-agent.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:8000 to use Argus Agent"
  kubectl port-forward --namespace {{ .Values.global.namespace }} $POD_NAME 8000:8000
{{- end }}

2. Check the status of the pods:
  kubectl get pods --namespace {{ .Values.global.namespace }} -l "app.kubernetes.io/instance={{ .Release.Name }}"

3. View logs:
  # API
  kubectl logs --namespace {{ .Values.global.namespace }} -l "app={{ .Values.api.name }}" --tail=100 -f

  # MCP Server
  kubectl logs --namespace {{ .Values.global.namespace }} -l "app={{ .Values.mcpServer.name }}" --tail=100 -f

  # UI
  kubectl logs --namespace {{ .Values.global.namespace }} -l "app={{ .Values.ui.name }}" --tail=100 -f

4. Access the services:
  # API
  kubectl port-forward --namespace {{ .Values.global.namespace }} svc/{{ include "argus-agent.fullname" . }}-api 8000:8000

  # MCP Server
  kubectl port-forward --namespace {{ .Values.global.namespace }} svc/{{ include "argus-agent.fullname" . }}-mcp-server 8002:8002

  # UI
  kubectl port-forward --namespace {{ .Values.global.namespace }} svc/{{ include "argus-agent.fullname" . }}-ui 8080:80

📚 Documentation: https://github.com/your-org/argus-mcp-agent
